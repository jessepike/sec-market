name: collect

# Twice-weekly collection of everything EDGAR's structured data can't reach —
# press releases, funding and acquisition announcements, state WARN notices,
# ratings actions, going-concern language, executive changes, divestitures,
# shutdowns, and each public supplier's own headline metric from its latest
# results release (autonomy-coverage-plan.md §5, "Everything unstructured").
# Runs headless Claude Code under the owner's own subscription login (never
# an API key), via scripts/collect.sh, which writes data files only and
# never touches git or the build gates itself. This workflow commits and
# pushes what that script produced, then hands the publish decision to
# scripts/auto-publish.sh exactly like edgar-refresh.yml does — nobody is
# asked (D-76). The prompt itself is docs/archive/2026-10-refresh-gate/auto-001/collect-brief.md,
# written and owned separately (step 3), so changing what the model is
# asked to do never means touching this workflow file or collect.sh.

on:
  schedule:
    - cron: '0 14 * * 2,5' # 14:00 UTC Tuesday and Friday
  workflow_dispatch:
    inputs:
      collect_only:
        description: 'Comma-separated supplier ids to restrict this run to (leave empty for every declared supplier)'
        required: false
        default: ''

permissions:
  contents: write

concurrency:
  group: auto-publish
  cancel-in-progress: false

jobs:
  collect:
    runs-on: ubuntu-latest
    timeout-minutes: 120
    steps:
      - uses: actions/checkout@v4
        with:
          ref: preview
          fetch-depth: 0

      - uses: actions/setup-node@v4
        with:
          node-version: '22'
          cache: npm
          cache-dependency-path: web/package-lock.json

      - name: Install dependencies
        working-directory: web
        run: npm ci

      - name: Install Claude Code
        run: npm install -g @anthropic-ai/claude-code@latest

      - name: Create the short-lived data branch
        id: branch
        run: |
          set -euo pipefail
          branch="data/collect-$(date -u +%Y%m%dT%H%M%SZ)"
          git checkout -B "$branch"
          echo "name=$branch" >> "$GITHUB_OUTPUT"

      - name: Run the collector (scripts/collect.sh)
        id: run
        env:
          # Subscription login token from `claude setup-token`, run once by
          # the owner on his own account — never an API key. See this
          # file's header for the one-time setup step.
          CLAUDE_CODE_OAUTH_TOKEN: ${{ secrets.CLAUDE_CODE_OAUTH_TOKEN }}
          SMP_MODEL: claude-sonnet-5
          SMP_COLLECT_ONLY: ${{ github.event.inputs.collect_only }}
        run: ./scripts/collect.sh

      - name: Read the run's state from the receipt
        id: receipt
        run: |
          set -euo pipefail
          state="$(node -p "require('./.collect/receipt.json').state")"
          echo "state=$state" >> "$GITHUB_OUTPUT"
          echo "### collect.sh receipt" >> "$GITHUB_STEP_SUMMARY"
          echo '```json' >> "$GITHUB_STEP_SUMMARY"
          cat .collect/receipt.json >> "$GITHUB_STEP_SUMMARY"
          echo '```' >> "$GITHUB_STEP_SUMMARY"

      - name: Stop here if the login is not valid
        if: steps.receipt.outputs.state == 'blocked-auth'
        run: |
          echo "::error::the Claude subscription login (CLAUDE_CODE_OAUTH_TOKEN) is not valid. Run 'claude setup-token' and update the repository secret."
          exit 1

      - name: Commit whatever scripts/collect.sh produced
        id: commit
        if: steps.receipt.outputs.state != 'blocked-auth'
        env:
          GIT_AUTHOR_NAME: Security Market Pulse bot
          GIT_AUTHOR_EMAIL: jesse@keatingpike.com
          GIT_COMMITTER_NAME: Security Market Pulse bot
          GIT_COMMITTER_EMAIL: jesse@keatingpike.com
        run: |
          set -euo pipefail
          git add \
            web/src/data/tracker/events.ts \
            web/src/data/tracker/suppliers.ts \
            web/src/data/evidence.ts \
            web/src/data/generated/collectState.json
          if git diff --cached --quiet; then
            echo "changed=false" >> "$GITHUB_OUTPUT"
            echo "collect: nothing to commit this run (state: ${{ steps.receipt.outputs.state }})"
          else
            state="${{ steps.receipt.outputs.state }}"
            git commit -m "data(collect): twice-weekly collection ($state)" \
              -m "$(cat .collect/receipt.json)"
            echo "changed=true" >> "$GITHUB_OUTPUT"
          fi

      - name: Push the branch
        if: steps.commit.outputs.changed == 'true'
        run: git push origin "${{ steps.branch.outputs.name }}"

      - name: Decide whether this publishes (scripts/auto-publish.sh)
        if: steps.commit.outputs.changed == 'true'
        env:
          RUNNER: collect
          BRANCH: ${{ steps.branch.outputs.name }}
          REMOTE: origin
          WEB_DIR: web
          GH_TOKEN: ${{ github.token }}
        run: ./scripts/auto-publish.sh
