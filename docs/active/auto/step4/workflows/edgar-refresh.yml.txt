name: edgar refresh

# Daily SEC EDGAR pull for public-company revenue, and its own auto-publish
# (autonomy-coverage-plan.md §5, step 2 collection + step 4 publishing). No
# model is involved in the collection itself: the script reads structured
# XBRL facts from data.sec.gov and computes quarters/YoY with plain
# arithmetic, so it cannot invent a number. Writes
# web/src/data/generated/edgarFacts.json and web/src/data/runStatus.ts only,
# on a short-lived branch, then hands the decision of whether any of that
# reaches `preview` or `main` to scripts/auto-publish.sh — nobody is asked
# (D-76).

on:
  schedule:
    - cron: '0 13 * * *' # 13:00 UTC daily
  workflow_dispatch:

permissions:
  contents: write

concurrency:
  group: auto-publish
  cancel-in-progress: false

jobs:
  refresh:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
        with:
          ref: preview
          fetch-depth: 0

      - uses: actions/setup-node@v4
        with:
          node-version: '22'
          cache: npm
          cache-dependency-path: web/package-lock.json

      - name: Install dependencies
        working-directory: web
        run: npm ci

      - name: Fetch EDGAR facts and compute quarters/YoY
        working-directory: web
        run: npm run refresh:edgar

      - name: Stop here if nothing changed
        id: diff
        run: |
          set -euo pipefail
          if git diff --quiet -- web/src/data/generated/edgarFacts.json; then
            echo "No change in edgarFacts.json today. Nothing to push."
            echo "changed=false" >> "$GITHUB_OUTPUT"
          else
            echo "changed=true" >> "$GITHUB_OUTPUT"
          fi

      # Written directly here, not in web/scripts/, because no script under
      # web/ owns "the EDGAR run's own honest status" today — the type only
      # (web/src/data/runStatus.ts) exists yet. A plain, self-contained edit,
      # same block-scoped approach scripts/auto-publish.sh's
      # push_incomplete_status uses for the failure path, so both paths
      # touch the file the same way.
      - name: Mark this run's status as updated
        if: steps.diff.outputs.changed == 'true'
        run: |
          set -euo pipefail
          node -e '
            const fs = require("fs");
            const path = "web/src/data/runStatus.ts";
            const today = new Date().toISOString().slice(0, 10);
            let src = fs.readFileSync(path, "utf8");
            const blockRe = /(export const runStatus: RunStatus = \{)([\s\S]*?)(\n\})/;
            const m = src.match(blockRe);
            if (!m) { console.error("edgar-refresh: runStatus.ts block shape not recognized"); process.exit(1); }
            let block = m[2];
            block = block.replace(/lastChecked: .*?,/, `lastChecked: ${JSON.stringify(today)},`);
            block = block.replace(/state: .*?,/, `state: "updated",`);
            block = block.replace(/failedSources: \[[\s\S]*?\],/, `failedSources: [],`);
            src = src.slice(0, m.index) + m[1] + block + m[3] + src.slice(m.index + m[0].length);
            fs.writeFileSync(path, src);
          '

      - name: Commit to a short-lived data branch and push it
        if: steps.diff.outputs.changed == 'true'
        id: branch
        env:
          GIT_AUTHOR_NAME: Security Market Pulse bot
          GIT_AUTHOR_EMAIL: jesse@keatingpike.com
          GIT_COMMITTER_NAME: Security Market Pulse bot
          GIT_COMMITTER_EMAIL: jesse@keatingpike.com
        run: |
          set -euo pipefail
          branch="data/edgar-$(date -u +%Y%m%dT%H%M%SZ)"
          git checkout -B "$branch"
          git add web/src/data/generated/edgarFacts.json web/src/data/runStatus.ts
          failures=$(node -p "require('./web/src/data/generated/edgarFacts.json').failures.length")
          total=$(node -p "require('./web/src/data/generated/edgarFacts.json').suppliers.length + require('./web/src/data/generated/edgarFacts.json').failures.length")
          git commit -m "data(edgar): refresh SEC EDGAR revenue facts ($failures/$total suppliers failed)"
          git push origin "$branch"
          echo "name=$branch" >> "$GITHUB_OUTPUT"

      - name: Decide whether this publishes (scripts/auto-publish.sh)
        if: steps.diff.outputs.changed == 'true'
        env:
          RUNNER: edgar
          BRANCH: ${{ steps.branch.outputs.name }}
          REMOTE: origin
          WEB_DIR: web
          GH_TOKEN: ${{ github.token }}
        run: ./scripts/auto-publish.sh
