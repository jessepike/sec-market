# Parse the actual workflow expression using its small explicit grammar.
# No GitHub workflow or remote service is invoked by these offline checks.
class Parser:
    def __init__(self,text):
        if text.startswith('${{'):
            assert text.endswith('}}');text=text[3:-2].strip()
        self.tokens=[];pos=0
        pattern=re.compile(r"\s*(\|\||&&|==|!=|[!()]|'[^']*'|[A-Za-z_][A-Za-z0-9_.-]*)")
        while pos<len(text):
            m=pattern.match(text,pos);assert m,(text,pos)
            self.tokens.append(m.group(1));pos=m.end()
        self.i=0
    def take(self,value=None):
        if self.i>=len(self.tokens):return None
        t=self.tokens[self.i]
        if value is not None and t!=value:return None
        self.i+=1;return t
    def atom(self):
        if self.take('!'):return ('not',self.atom())
        if self.take('('):
            n=self.or_expr();assert self.take(')');return n
        token=self.take();assert token
        if token.startswith("'"):return ('literal',token[1:-1])
        if token in ['true','false']:return ('literal',token=='true')
        if self.take('('):
            assert self.take(')');assert token in ['always','cancelled'];return ('status',token)
        return ('value',token)
    def equality(self):
        node=self.atom()
        if self.i<len(self.tokens) and self.tokens[self.i] in ['==','!=']:
            op=self.take();node=(op,node,self.atom())
        return node
    def and_expr(self):
        node=self.equality()
        while self.take('&&'):node=('and',node,self.equality())
        return node
    def or_expr(self):
        node=self.and_expr()
        while self.take('||'):node=('or',node,self.and_expr())
        return node
    def parse(self):
        result=self.or_expr();assert self.i==len(self.tokens);return result

def evaluate(n,context):
    op=n[0]
    if op=='literal':return n[1]
    if op=='value':return context[n[1]]
    if op=='status':return True if n[1]=='always' else context['cancelled']
    if op=='not':return not evaluate(n[1],context)
    if op=='and':return bool(evaluate(n[1],context) and evaluate(n[2],context))
    if op=='or':return bool(evaluate(n[1],context) or evaluate(n[2],context))
    if op in ['==','!=']:
        left,right=evaluate(n[1],context),evaluate(n[2],context)
        # The declared contexts use Boolean private flags and string outputs.
        # GitHub compares strings without regard to case; no mixed types here.
        if isinstance(left,str) and isinstance(right,str):left,right=left.casefold(),right.casefold()
        equal=left==right
        return equal if op=='==' else not equal
    raise AssertionError(op)
