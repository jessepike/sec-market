class Timed(unittest.TextTestResult):
    def startTest(self,test):
        self.begin=time.monotonic();self.cpu=time.process_time();self.before=counts.copy()
        print(json.dumps({'event':'start','test':test.id(),'atMonotonic':self.begin}),flush=True)
        return super().startTest(test)
    def stopTest(self,test):
        row={'test':test.id(),'wallSeconds':time.monotonic()-self.begin,'cpuSeconds':time.process_time()-self.cpu,'counters':dict(counts-self.before)}
        rows.append(row);print(json.dumps({'event':'stop',**row}),flush=True)
        (OUT/'per-test-progress.json').write_text(json.dumps(rows,indent=2)+'\n')
        return super().stopTest(test)

def wrap(obj,name):
    original=getattr(obj,name)
    def measured(*a,**kw):
        counts[name+'Calls']+=1
        start=time.process_time()
        try:return original(*a,**kw)
        finally:counts[name+'CpuSeconds']+=time.process_time()-start
    return patch.object(obj,name,measured)
